// translations (10 items/language, 2/category x 5 categories).
//
// ----------------------------------------------------------------------------
// CRITICAL — PageBreakItem.setGoToPage() semantics (gotten wrong twice
// before this version, so spelled out in full here):
//
//   somePageBreak.setGoToPage(target) sets the navigation for the SECTION
//   THAT ENDS immediately before somePageBreak — i.e. it controls where the
//   PREVIOUS page's "Next" button goes, NOT where somePageBreak's own page
//   goes.
//
//   So: to control where page P exits to, you call .setGoToPage() on
//   whichever page-break STARTS THE NEXT page after P in document order —
//   never on the page-break that starts P itself.
//
//   The setExitOfPageBefore() helper below exists purely so every call site
//   reads unambiguously ("this line closes the page BEFORE this break"),
//   instead of relying on remembering the off-by-one rule each time.
// ----------------------------------------------------------------------------
//
// FLOW:
//   Page 0  Intro (English) + age gate -> "No" -> age-exit -> SUBMIT
//   Page 1  Native-language question (English): DE/FR/ES/AR/None
//     "None of these" -> none-exit (includes Image Compliance link) -> SUBMIT
//     DE/FR/ES/AR -> that language's consent page
//   Per language, back to back: consent -> (Yes) items -> feedback -> SUBMIT
//                                        -> (No)  that language's decline -> SUBMIT
//   All four decline pages, then age-exit, then none-exit are grouped at the
//   very end, each with an explicit SUBMIT exit — nothing left to the
//   document-order default anywhere in the whole form.
//
// Each rated item displays the English original above its translation.
// Item order is NOT shuffled — it comes straight from Data_NativeSpeaker.gs,
// which is itself in a fixed, deterministic order (see
// phase2b_native_speaker_formdata.py).
// ============================================================================

// Reads unambiguously at every call site: "this closes the page BEFORE
// nextPageBreak". See the CRITICAL note at the top of this file.
function setExitOfPageBefore(nextPageBreak, target) {
  nextPageBreak.setGoToPage(target);
}

function generateForm() {
  if (typeof BRIEF_DATA === "undefined") {
    Logger.log("STOPPED — Data_NativeSpeaker script file not found. Create a " +
      "file named exactly 'Data_NativeSpeaker' and paste " +
      "Data_NativeSpeaker.gs.txt's content into it, then run this again.");
    return;
  }

  var form = FormApp.create(EN.introTitle);
  form.setCollectEmail(false);
  form.setLimitOneResponsePerUser(false);
  form.setShowLinkToRespondAgain(false);
  form.setProgressBar(false);
  form.setConfirmationMessage(EN.closingText);

  // ---- Page 0: intro + age gate ----
  form.addSectionHeaderItem().setTitle(EN.introHeading).setHelpText(EN.introBody);
  var ageItem = form.addMultipleChoiceItem().setTitle(EN.ageQuestion).setRequired(true);

  // ---- Page 1: native-language question ----
  form.addPageBreakItem().setTitle("Getting Started");
  var nativeItem = form.addMultipleChoiceItem().setTitle(EN.nativeLanguageQuestion).setRequired(true);

  // ---- Per language, back to back: consent -> items -> feedback ----
  var consentItems = {};       // lang -> MultipleChoiceItem (for wiring "No")
  var languageEntryPages = {}; // lang -> consent PageBreakItem (native-language branch target)
  var itemsPages = {};         // lang -> items PageBreakItem (consent "Yes" branch target)

  LANGUAGE_KEYS.forEach(function (lang) {
    var meta = LANG_META[lang];
    var items = BRIEF_DATA[lang]; // fixed, deterministic order — no shuffle

    // Consent page
    var pageConsent = form.addPageBreakItem().setTitle(meta.label);
    languageEntryPages[lang] = pageConsent;
    form.addSectionHeaderItem().setTitle(meta.consentTitle).setHelpText(meta.consentText);
    var consentItem = form.addMultipleChoiceItem().setTitle(meta.consentQuestion).setRequired(true);
    consentItems[lang] = consentItem;
    // Consent's own exit is fully choice-driven (Yes -> items, No -> decline,
    // wired further below) — no setGoToPage() needed on any page-break here.

    // Items page — all 10 rating items on one page.
    itemsPages[lang] = form.addPageBreakItem().setTitle(meta.label);
    form.addSectionHeaderItem().setTitle(meta.consentTitle)
      .setHelpText(meta.ratingInstruction + " (1 = " + meta.scaleLow + ", 5 = " + meta.scaleHigh + ")");
    items.forEach(function (item) {
      var combinedText = meta.englishLabel + ":\n" + item.text_en +
        "\n\n" + meta.translationLabel + ":\n" + item.text;
      form.addGridItem()
        .setTitle(item.brief_id)
        .setHelpText(combinedText)
        .setRows([meta.fluencyLabel, meta.adequacyLabel])
        .setColumns(["1", "2", "3", "4", "5"])
        .setRequired(true);
    });
    // Items page's own exit: we WANT the default "continue to next section"
    // here, since the next page-break added is this same language's
    // feedback page — so deliberately no setGoToPage() call on it.

    // Feedback page — separate page, not folded under the items.
    form.addPageBreakItem().setTitle(EN.feedbackTitle);
    form.addParagraphTextItem().setTitle(EN.feedbackQuestion).setRequired(false);
    // Feedback page's own exit is set below, once the NEXT page-break in
    // document order exists (either the next language's consent page, or —
    // for Arabic, the last language — the first decline page).
  });

  // ---- Decline-exit pages, grouped together after all four languages ----
  var declineExitPages = {};
  LANGUAGE_KEYS.forEach(function (lang) {
    var meta = LANG_META[lang];
    var pageDecline = form.addPageBreakItem().setTitle(meta.declineTitle);
    declineExitPages[lang] = pageDecline;
    form.addSectionHeaderItem().setTitle(meta.declineText); // message only, no title repeat
  });

  // ---- Age-exit and none-exit, last two pages in the document ----
  var pageAgeExit = form.addPageBreakItem().setTitle(EN.ageExitTitle);
  form.addSectionHeaderItem().setTitle(EN.ageExitText); // message only, no title repeat

  var pageNoneExit = form.addPageBreakItem().setTitle(EN.noneExitTitle);
  form.addSectionHeaderItem().setTitle(EN.noneExitText + "\n\n" + EN.noneExitLinkLabel + IMAGE_SURVEY_LINK);

  // ---- Wire branch choices (reliable regardless of document distance) ----
  ageItem.setChoices([
    ageItem.createChoice(EN.ageOptions[0], FormApp.PageNavigationType.CONTINUE),
    ageItem.createChoice(EN.ageOptions[1], pageAgeExit)
  ]);

  var nativeChoices = LANGUAGE_KEYS.map(function (lang) {
    return nativeItem.createChoice(EN.nativeLanguageOptions[lang], languageEntryPages[lang]);
  });
  nativeChoices.push(nativeItem.createChoice(EN.nativeLanguageOptions.none, pageNoneExit));
  nativeItem.setChoices(nativeChoices);

  LANGUAGE_KEYS.forEach(function (lang) {
    var meta = LANG_META[lang];
    consentItems[lang].setChoices([
      consentItems[lang].createChoice(meta.consentOptions[0], itemsPages[lang]),
      consentItems[lang].createChoice(meta.consentOptions[1], declineExitPages[lang])
    ]);
  });

  // ---- Wire every default-flow exit EXPLICITLY, using setExitOfPageBefore ----
  // Document order after the per-language blocks is:
  //   [DE consent][DE items][DE feedback]
  //   [FR consent][FR items][FR feedback]
  //   [ES consent][ES items][ES feedback]
  //   [AR consent][AR items][AR feedback]
  //   [DE decline][FR decline][ES decline][AR decline]
  //   [age-exit][none-exit]
  //
  // Each language's OWN feedback page must end in SUBMIT, not "continue"
  // into the next language's consent page (the document-order default).
  // Per the rule at the top of this file, that means calling setGoToPage()
  // on the PageBreakItem that STARTS the page immediately after it.
  setExitOfPageBefore(languageEntryPages["fr"], FormApp.PageNavigationType.SUBMIT); // closes DE feedback
  setExitOfPageBefore(languageEntryPages["es"], FormApp.PageNavigationType.SUBMIT); // closes FR feedback
  setExitOfPageBefore(languageEntryPages["ar"], FormApp.PageNavigationType.SUBMIT); // closes ES feedback
  setExitOfPageBefore(declineExitPages["de"], FormApp.PageNavigationType.SUBMIT);   // closes AR feedback

  // Each decline page must also end in SUBMIT immediately, not continue on
  // into the next language's decline page.
  setExitOfPageBefore(declineExitPages["fr"], FormApp.PageNavigationType.SUBMIT); // closes DE decline
  setExitOfPageBefore(declineExitPages["es"], FormApp.PageNavigationType.SUBMIT); // closes FR decline
  setExitOfPageBefore(declineExitPages["ar"], FormApp.PageNavigationType.SUBMIT); // closes ES decline
  setExitOfPageBefore(pageAgeExit, FormApp.PageNavigationType.SUBMIT);            // closes AR decline
  setExitOfPageBefore(pageNoneExit, FormApp.PageNavigationType.SUBMIT);           // closes age-exit
  // none-exit is the LAST page in the document — nothing follows it, so its
  // own exit is the form's natural end (Submit) with no call needed.

  var url = form.getEditUrl() + "  |  PUBLISHED: " + form.getPublishedUrl();
  Logger.log(url);
  return url;
}
